function ct(name, over = {}) {
  const t = CUMULUS.type[name] || CUMULUS.type.body;
  return {
    fontFamily: CUMULUS.font[t.family],
    fontSize: t.size,
    lineHeight: t.line + 'px',
    fontWeight: t.weight,
    letterSpacing: t.tracking + 'px',
    textTransform: t.upper ? 'uppercase' : 'none',
    margin: 0,
    ...over,
  };
}

window.CUMULUS = CUMULUS;
window.ct = ct;
